## 다시 만들기
```bash
cd videos/olds-diploma-longform
python3 -I tools/tts.py            # 음성 mp3가 있으면 재생성 없이 wav/타이밍만 다시 만듦 (--force로 재생성)

python3 -I tools/sfx_synth.py      # 효과음 합성
python3 -I tools/build.py && npx --yes hyperframes@0.8.174 check
npx --yes hyperframes@0.8.174 render -o renders/video_hf.mp4 --fps 30
python3 -I tools/mix.py renders/video_hf.mp4   # -> renders/video.mp4, renders/loudness.json
```
